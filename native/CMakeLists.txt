cmake_minimum_required(VERSION 3.20)
project(JackCodexProbe LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Only Windows x64 / J.A.C.K. 1.1.3773 Freeware is validated")
endif()
add_library(codex_bridge MODULE bridge.cpp)
target_compile_features(codex_bridge PRIVATE cxx_std_17)
target_compile_definitions(codex_bridge PRIVATE WIN32 JACK_64BIT JACK_API_VERSION=10
  ENTITYAPI_NO_INLINEFUNCS NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(codex_bridge PRIVATE vendor/JackSDK/sdk/public vendor/json)
target_link_libraries(codex_bridge PRIVATE bcrypt advapi32 user32)
set_target_properties(codex_bridge PROPERTIES OUTPUT_NAME vpCodexBridgex64
  MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_options(codex_bridge PRIVATE /W4 /EHsc /guard:cf)
target_link_options(codex_bridge PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
add_executable(capture_contract capture_contract.cpp)
target_compile_features(capture_contract PRIVATE cxx_std_17)
target_compile_definitions(capture_contract PRIVATE WIN32 JACK_64BIT JACK_API_VERSION=10 ENTITYAPI_NO_INLINEFUNCS NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(capture_contract PRIVATE vendor/JackSDK/sdk/public vendor/json)
target_link_libraries(capture_contract PRIVATE bcrypt advapi32 user32)
enable_testing()
add_test(NAME native_capture_contract COMMAND capture_contract)
